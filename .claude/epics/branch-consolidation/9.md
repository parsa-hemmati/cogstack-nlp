class FHIRMapper:
    """Maps MedCAT NLP output to FHIR R4 resources."""

    def map_entities_to_observations(
        self,
        entities: List[MedCATEntity],
        patient_id: str
    ) -> List[Observation]:
        """Convert MedCAT entities to FHIR Observations.

        Critical: Only map entities with appropriate meta-annotations:
        - Negation: Affirmed (not negated)
        - Experiencer: Patient (not family/other)
        - Temporality: Current/Recent (not historical)
        """
        observations = []
        for entity in entities:
            # Apply meta-annotation filtering
            if self._is_active_patient_condition(entity):
                obs = self._build_observation(entity, patient_id)
                observations.append(obs)
        return observations

    def _is_active_patient_condition(self, entity: MedCATEntity) -> bool:
        """Validate meta-annotations for clinical relevance."""
        meta = entity.meta_anns
        return (
            meta.get('Negation') == 'Affirmed' and
            meta.get('Experiencer') == 'Patient' and
            meta.get('Temporality') in ['Current', 'Recent']
        )
```

class ICD10Coder:
    """Maps clinical concepts to ICD-10-CM codes."""

    def map_snomed_to_icd10(
        self,
        snomed_code: str,
        context: Optional[str] = None
    ) -> List[ICD10Code]:
        """Map SNOMED-CT code to ICD-10-CM codes.

        Uses SNOMED-to-ICD-10 mappings from UMLS.
        Returns multiple codes if applicable (e.g., laterality, severity).
        """
        # Query UMLS mapping table
        mappings = self.umls_client.get_mappings(
            source_code=snomed_code,
            source_vocabulary='SNOMEDCT_US',
            target_vocabulary='ICD10CM'
        )

        # Rank by confidence and context relevance
        codes = self._rank_and_filter(mappings, context)
        return codes

    def validate_code(self, icd10_code: str) -> ValidationResult:
        """Validate ICD-10 code format and existence."""
        # Check format (e.g., E11.9, J44.0)
        if not self._is_valid_format(icd10_code):
            return ValidationResult(valid=False, error="Invalid format")

        # Check existence in ICD-10-CM catalog
        if not self._code_exists(icd10_code):
            return ValidationResult(valid=False, error="Code not found")

        return ValidationResult(valid=True)
```

class SafetyChecker:
    """Validates clinical data for patient safety issues."""

    def check_critical_results(
        self,
        observations: List[Observation]
    ) -> List[SafetyAlert]:
        """Identify critical lab results or vital signs.

        Examples:
        - Potassium <2.5 or >6.0 mEq/L (critical)
        - Glucose <40 or >500 mg/dL (critical)
        - Systolic BP >180 mmHg (critical)
        """
        alerts = []
        for obs in observations:
            if self._is_critical_value(obs):
                alert = SafetyAlert(
                    severity='CRITICAL',
                    message=f"Critical {obs.code.text}: {obs.value}",
                    observation_id=obs.id,
                    action_required=True
                )
                alerts.append(alert)
        return alerts

    def check_drug_interactions(
        self,
        medications: List[Medication]
    ) -> List[SafetyAlert]:
        """Identify potential drug-drug interactions."""
        alerts = []
        for med1, med2 in combinations(medications, 2):
            interaction = self.drug_db.check_interaction(med1, med2)
            if interaction.severity in ['high', 'critical']:
                alerts.append(SafetyAlert(
                    severity=interaction.severity.upper(),
                    message=interaction.description,
                    medications=[med1.code, med2.code]
                ))
        return alerts
```

class CohortIdentifier:
    """Identifies patient cohorts based on clinical criteria."""

    def identify_cohort(
        self,
        criteria: CohortCriteria
    ) -> List[Patient]:
        """Find patients matching clinical criteria.

        Example:
        - Criteria: Type 2 Diabetes + HbA1c >9% + Age 40-65
        - Uses: MedCAT for condition extraction, meta-annotations for filtering
        """
        # Build Elasticsearch query
        query = self._build_cohort_query(criteria)

        # Execute search
        results = self.search_service.search_patients(query)

        # Apply meta-annotation filters (critical for accuracy)
        filtered_results = [
            patient for patient in results
            if self._matches_criteria_with_meta_anns(patient, criteria)
        ]

        return filtered_results

    def _matches_criteria_with_meta_anns(
        self,
        patient: Patient,
        criteria: CohortCriteria
    ) -> bool:
        """Validate patient matches criteria using meta-annotations."""
        # Get patient conditions
        conditions = patient.conditions

        # Apply meta-annotation filters
        active_conditions = [
            c for c in conditions
            if c.meta_anns.get('Negation') == 'Affirmed' and
               c.meta_anns.get('Experiencer') == 'Patient' and
               c.meta_anns.get('Temporality') in ['Current', 'Recent']
        ]

        # Check if criteria met
        return criteria.matches(active_conditions)
```
