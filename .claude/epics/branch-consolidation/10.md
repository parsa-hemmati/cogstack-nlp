@router.post("/patients/search", response_model=PatientSearchResponse)
async def search_patients(
    query: PatientSearchQuery,
    user: User = Depends(get_current_user),
    search_service: PatientSearchService = Depends()
) -> PatientSearchResponse:
    """Search for patients matching clinical criteria.

    Consolidates:
    - Basic search (search-optimization)
    - Meta-annotation filtering (medcat-integration)
    - Performance optimizations (api-optimization)
    """
    # Audit log PHI access
    await audit_logger.log_patient_search(user.id, query)

    # Execute search with caching
    cache_key = f"search:{hash(query)}"
    cached = await cache.get(cache_key)
    if cached:
        return PatientSearchResponse.parse_raw(cached)

    # Search with meta-annotation filtering
    results = await search_service.search(query)

    # Cache results (5 min TTL)
    await cache.set(cache_key, results.json(), ttl=300)

    return results

**MedCAT Annotation** (`app/api/v1/medcat.py`):
```python
# Branches: medcat-integration, api-optimization
@router.post("/medcat/annotate", response_model=AnnotationResponse)
async def annotate_text(
    request: AnnotationRequest,
    user: User = Depends(get_current_user),
    medcat_service: MedCATService = Depends(),
    background_tasks: BackgroundTasks
) -> AnnotationResponse:
    """Annotate medical text with MedCAT.

    Consolidates:
    - Synchronous annotation (medcat-integration)
    - Async annotation with background tasks (api-optimization)
    - Meta-annotation extraction (medcat-integration)
    """
    # For short text, annotate synchronously
    if len(request.text) < 5000:
        annotations = await medcat_service.annotate(request.text)
        return AnnotationResponse(
            annotations=annotations,
            status="completed"
        )

    # For long text, queue background task
    task_id = await medcat_service.queue_annotation(request.text)
    background_tasks.add_task(
        process_annotation_task,
        task_id,
        request.text
    )

    return AnnotationResponse(
        task_id=task_id,
        status="queued"
    )
```

@router.post("/safety/check", response_model=SafetyCheckResponse)
async def check_safety(
    request: SafetyCheckRequest,
    user: User = Depends(get_current_user),
    safety_service: SafetyService = Depends()
) -> SafetyCheckResponse:
    """Check observations for patient safety issues.

    Consolidates:
    - Safety checker (patient-safety-ui)
    - Parallel processing (api-optimization)
    """
    # Run safety checks in parallel
    critical_values, drug_interactions, guideline_violations = await asyncio.gather(
        safety_service.check_critical_values(request.observations),
        safety_service.check_drug_interactions(request.medications),
        safety_service.check_guidelines(request.conditions)
    )

    # Consolidate alerts
    alerts = critical_values + drug_interactions + guideline_violations

    # Sort by severity
    alerts.sort(key=lambda a: a.severity_score, reverse=True)

    return SafetyCheckResponse(alerts=alerts)

#### 4. Caching Strategy Consolidation

**Redis Cache** (from `api-optimization`):
```python
# app/core/cache.py
class CacheService:
    """Unified caching service using Redis."""

    def __init__(self):
        self.redis = Redis.from_url(settings.REDIS_URL)

    async def get(self, key: str) -> Optional[str]:
        """Get cached value."""
        return await self.redis.get(key)

    async def set(self, key: str, value: str, ttl: int = 300):
        """Set cached value with TTL."""
        await self.redis.setex(key, ttl, value)

    async def invalidate(self, pattern: str):
        """Invalidate cache keys matching pattern."""
        keys = await self.redis.keys(pattern)
        if keys:
            await self.redis.delete(*keys)
```

**Cache usage patterns:**
- Search results: 5 min TTL
- SNOMED→ICD-10 mappings: 1 hour TTL
- Patient data: No cache (always fresh, PHI concern)
- Static data (code systems): 24 hour TTL

**In-memory cache** (from `api-optimization`):
- LRU cache for frequently accessed data
